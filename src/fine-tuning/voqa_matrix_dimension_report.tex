\begin{abstract}
This technical report delivers a rigorous, scratch-level mathematical dissection of the \textbf{InternVL 2.5-1B} vision-language model deployed across the Multilingual Visual-only Question Answering (\textbf{VoQA}) pipeline. We trace the exact tensor dimensional flow from raw pixel grids ($448 \times 448$) through spatial downsampling, multimodal linear projection, multi-head causal self-attention, SwiGLU MLP compression/decompression, and language generation heads. Furthermore, we provide a complete matrix taxonomy identifying all $315$ base parameter matrices, detail the mechanics of Low-Rank Adaptation (LoRA), and formally verify the parameter isolation between Stage 1 (Universal Task Adapters) and Stage 3 (Multilingual LoRA Stacking).
\end{abstract}

\section{LoRA Attachment Topology: Stage 1 vs. Stage 3}
% ===========================================================================

Low-Rank Adaptation modifies a frozen linear projection $W_0 \in \mathbb{R}^{d_{\text{out}} \times d_{\text{in}}}$ by introducing an intrinsic low-rank decomposition:
\begin{equation}
W = W_0 + \Delta W = W_0 + \frac{\alpha}{r} (B \cdot A)
\end{equation}
where $A \in \mathbb{R}^{r \times d_{\text{in}}}$ is initialized from a Gaussian distribution $\mathcal{N}(0, \sigma^2)$, $B \in \mathbb{R}^{d_{\text{out}} \times r}$ is initialized to zero, and $r \ll \min(d_{\text{in}}, d_{\text{out}})$.

Table~\ref{tab:lora_topology} delineates all possible theoretical adapter injection sites versus the empirical activation state across Pipeline Stages 1 and 3.

\section{Step-by-Step Dimensional Trace on VoQA Benchmark Sample}
% ===========================================================================
We trace a concrete multimodal forward pass executing on the B200 cluster:
\begin{itemize}
    \item \textbf{Image:} 1 airport tarmac scene with rendered watermark question (\texttt{448 $\times$ 448} pixels).
    \item \textbf{Prompt:} \texttt{"USER: <image>\textbackslash nProvide a one-sentence caption... Reference OCR token: FawsaG ASSISTANT: People are boarding..."} ($50$ text tokens).
\end{itemize}

\subsection{Phase 1: Vision Tower Internal Dynamics}
\begin{enumerate}
    \item \textbf{Input Image:} $X_0 \in \mathbb{R}^{1 \times 3 \times 448 \times 448}$.
    \item \textbf{Patch Extraction:} Kernel $14 \times 14$, Stride $14 \implies (448/14) \times (448/14) = 32 \times 32 = \mathbf{1024\text{ patches}}$.
    \item \textbf{Patch Embedding:} Convolved to $1024\text{d} \implies X_{\text{patch}} \in \mathbb{R}^{1 \times 1024 \times 1024}$.
    \item \textbf{ViT Layer Attention ($l \in [1, 24]$):}
    \begin{align*}
        Q, K, V &= X \cdot W_Q, \quad X \cdot W_K, \quad X \cdot W_V \quad \in \mathbb{R}^{1 \times 1024 \times 1024} \\
        \text{AttnScores} &= \text{Softmax}\left(\frac{Q K^T}{\sqrt{64}}\right) \in \mathbb{R}^{1 \times 16 \times 1024 \times 1024} \\
        X_{\text{attn}} &= X + (\text{AttnScores} \cdot V) \cdot W_O \quad \in \mathbb{R}^{1 \times 1024 \times 1024}
    \end{align*}
    \item \textbf{ViT Layer MLP (Expansion \& Compression):}
    \begin{align*}
        H_{\text{exp}} &= \text{GELU}(X_{\text{attn}} \cdot W_{\text{fc1}}) \in \mathbb{R}^{1 \times 1024 \times 4096} \quad (\mathbf{4\times \text{ Decompression}}) \\
        X_{\text{out}} &= X_{\text{attn}} + (H_{\text{exp}} \cdot W_{\text{fc2}}) \in \mathbb{R}^{1 \times 1024 \times 1024} \quad (\mathbf{\text{Compression to } 1024})
    \end{align*}
    \item \textbf{Spatial Pixel-Shuffle Downsampling (2$\times$2 Merging):}
    The $32 \times 32$ spatial patch layout is grouped into $16 \times 16$ macro-blocks:
    \begin{equation}
        \frac{32 \times 32}{2 \times 2} = \mathbf{256\text{ visual tokens}} \implies X_{\text{vit\_final}} \in \mathbb{R}^{1 \times 256 \times 1024}
    \end{equation}
\end{enumerate}

\subsection{Phase 2: Multimodal Connector (Stage 1 Active Site)}
\begin{equation}
    X_{\text{visual}} = (X_{\text{vit\_final}} \cdot W_{\text{connector}}^T) + b_{\text{connector}} \in \mathbb{R}^{1 \times 256 \times 896}
\end{equation}
This projection aligns the $1024$-dimensional visual representations into the $896$-dimensional Qwen2 manifold.

\subsection{Phase 3: Multimodal Sequence Merging}
The text tokens $T_{\text{ids}} \in \mathbb{Z}^{1 \times 50}$ pass through \texttt{embed\_tokens} to yield $T_{\text{embed}} \in \mathbb{R}^{1 \times 50 \times 896}$. The placeholder token \texttt{<image>} ($1$ token) is replaced with the $256$ visual tokens:
\begin{equation}
    S = 256\ (\text{visual}) + 50\ (\text{text}) = \mathbf{306\text{ tokens}} \implies X_{\text{merged}} \in \mathbb{R}^{1 \times 306 \times 896}
\end{equation}

\subsection{Phase 4: Qwen2 Layer Dynamics with Stage 3 LoRA Injections}
Across every layer $l \in [1, 24]$, the tensor $X \in \mathbb{R}^{1 \times 306 \times 896}$ undergoes:
\begin{enumerate}
    \item \textbf{LoRA Query Projection:}
    \begin{equation}
        Q = X W_Q + \frac{\alpha}{r} (X A_Q B_Q) \quad \text{where } A_Q \in \mathbb{R}^{896 \times 8},\ B_Q \in \mathbb{R}^{8 \times 896} \implies Q \in \mathbb{R}^{1 \times 306 \times 896}
    \end{equation}
    \item \textbf{LoRA Key \& Value Projections (Grouped-Query Attention):}
    \begin{align*}
        K &= X W_K + \frac{\alpha}{r} (X A_K B_K) \in \mathbb{R}^{1 \times 306 \times 128} \quad (\text{2 KV heads} \times 64\text{d}) \\
        V &= X W_V + \frac{\alpha}{r} (X A_V B_V) \in \mathbb{R}^{1 \times 306 \times 128}
    \end{align*}
    \item \textbf{Causal Multimodal Self-Attention:}
    Every textual token attends back to all $256$ visual feature vectors:
    \begin{equation}
        \text{AttnMap} = \text{Softmax}\left(\frac{Q K^T}{\sqrt{64}} + M_{\text{causal}}\right) \in \mathbb{R}^{1 \times 14 \times 306 \times 306}
    \end{equation}
    \item \textbf{LoRA SwiGLU Feed-Forward Network:}
    \begin{align*}
        \text{Gate} &= X W_{\text{gate}} + \frac{\alpha}{r}(X A_{\text{gate}} B_{\text{gate}}) \in \mathbb{R}^{1 \times 306 \times 4864} \quad (\mathbf{\text{Expansion to } 4864}) \\
        \text{Up} &= X W_{\text{up}} + \frac{\alpha}{r}(X A_{\text{up}} B_{\text{up}}) \in \mathbb{R}^{1 \times 306 \times 4864} \quad (\mathbf{\text{Expansion to } 4864}) \\
        H_{\text{swiglu}} &= \text{SiLU}(\text{Gate}) \odot \text{Up} \in \mathbb{R}^{1 \times 306 \times 4864} \\
        X_{\text{out}} &= X + \left[ H_{\text{swiglu}} W_{\text{down}} + \frac{\alpha}{r}(H_{\text{swiglu}} A_{\text{down}} B_{\text{down}}) \right] \in \mathbb{R}^{1 \times 306 \times 896} \quad (\mathbf{\text{Compression to } 896})
    \end{align*}
\end{enumerate}

\subsection{Phase 5: Output Logits \& Loss Formulation}
\begin{equation}
    \text{Logits} = X_{\text{out}}^{(24)} \cdot W_{\text{lm\_head}}^T \in \mathbb{R}^{1 \times 306 \times 151674}
\end{equation}
Cross-entropy loss is evaluated strictly against the non-masked answer token positions:
\begin{equation}
    \mathcal{L} = -\frac{1}{N_{\text{answer}}} \sum_{i \in \text{Answer}} \log P(y_i \mid y_{<i}, X_{\text{visual}}) = 0.8524
\end{equation}
In \textbf{Stage 1}, backpropagation computes $\nabla_{W_{\text{connector}}} \mathcal{L}$ to optimize the 918,400 connector parameters. In \textbf{Stage 3}, backpropagation computes $\nabla_{A, B} \mathcal{L}$ updating strictly the 336 LoRA micro-matrices.
